# ── MONITOR ───────────────────────────────────────────────────────────────────
# Live logs
journalctl -u beast-lbank-monitor -f

# Live logs filtered (signals, trades, errors)
journalctl -u beast-lbank-monitor -f | grep -E "signal|trade|error|slope|macro|entry|exit"

# Last 50 lines
journalctl -u beast-lbank-monitor -n 50

# Last 50 lines filtered
journalctl -u beast-lbank-monitor -n 50 --no-pager

# ── SERVICE ───────────────────────────────────────────────────────────────────
# Restart monitor
systemctl restart beast-lbank-monitor

# Restart API
systemctl restart beast-lbank-api

# Restart both
systemctl restart beast-lbank-monitor beast-lbank-api

# Status
systemctl status beast-lbank-monitor
systemctl status beast-lbank-api

# Stop / Start
systemctl stop beast-lbank-monitor
systemctl start beast-lbank-monitor

# Open trades

# Recent closed trades
sqlite3 /opt/beast-lbank/trades.db "SELECT id, symbol, side, close_reason, pnl_usd FROM trades WHERE stage = 'closed' ORDER BY closed_at DESC LIMIT 20;"

# Pending signals
sqlite3 /opt/beast-lbank/trades.db "SELECT id, symbol, action, status, created_at FROM signals WHERE status IN ('pending_eval','pending_memory') ORDER BY created_at DESC;"

# Recent signals
sqlite3 /opt/beast-lbank/trades.db "SELECT id, symbol, action, status, rejection_reason FROM signals ORDER BY created_at DESC LIMIT 20;"
